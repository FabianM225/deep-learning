# ============================================================
# scAAnet requirements
# ============================================================
#
# IMPORTANT — Python version:
#   Use Python >= 3.9 and <= 3.11.
#   TensorFlow 2 does NOT officially support Python 3.12 or 3.13.
#   Keep this in a dedicated env separate from the rest of the project.
#
# Recommended setup:
#   conda create -n scaanet python=3.11
#   conda activate scaanet
#   pip install -r requirements_scaanet.txt
#
# ============================================================
# GPU support on Windows
# ============================================================
#
#   TF 2.10 is the LAST version with native Windows GPU support.
#   TF 2.11+ requires WSL2 for GPU on Windows.
#
#   TF 2.10 requires:
#     - CUDA 11.2   (https://developer.nvidia.com/cuda-11.2.0-download-archive)
#     - cuDNN 8.1   (https://developer.nvidia.com/rdp/cudnn-archive)
#   Install both from NVIDIA before running — pip cannot install them.
#
#   WARNING: PyTorch in the main project typically ships with CUDA 12.x.
#   Running TF 2.10 (CUDA 11.2) alongside PyTorch (CUDA 12.x) in the same
#   Python environment can cause DLL conflicts on Windows. Use separate envs.
#
#   To verify GPU is visible after install:
#     python -c "import tensorflow as tf; print(tf.config.list_physical_devices('GPU'))"
#
#   On Linux / WSL2 you can use newer TF without the version pin below:
#     tensorflow[and-cuda]>=2.13.0,<2.16.0
# ============================================================
